Exiting.\n"; exit(0); } return trim(s); } // Reads an int in [lo, hi]; repeats until valid. int readInt(const string &prompt, int lo, int hi) { while (true) { string s = readLine(prompt); try { size_t pos = 0; int v = stoi(s, &pos); if (pos == s.size() && v >= lo && v <= hi) return v; } catch (...) {} cout << " Invalid input! Enter a number between " << lo << " and " << hi << ".\n"; } } string stars(int r) { if (r <= 0) return "not rated"; return string(r, '*') + string(5 - r, '-'); } string clip(const string &s, size_t n) { return s.size() <= n ? s : s.substr(0, n - 3) + "..."; } void line(char c = '-', int n = 78) { cout << string(n, c) << "\n"; } } // namespace util // ---------------------------------------------------------------- manager class FeedbackManager { vector records; int nextId = 1001; static vector seed() { return { {1001, "Hostel", "Water cooler on 2nd floor of Hostel J not working for a week.", 2, "In Progress"}, {1002, "Canteen", "Food quality in the canteen has dropped, rice often undercooked.", 2, "Pending"}, {1003, "Academics", "Lab sessions clash with lecture timings on Wednesday.", 3, "Resolved"}, {1004, "Infrastructure", "Wi-Fi keeps disconnecting in the library reading hall.", 1, "Pending"}, {1005, "Hostel", "Hostel Wi-Fi is very slow after 10 pm.", 2, "Pending"}, {1006, "Canteen", "Canteen should add more vegetarian options.", 4, "Resolved"}, {1007, "Faculty", "Please share lecture slides before class.", 0, "Pending"}, {1008, "Hostel", "Water leakage in the bathroom near room 214.", 1, "In Progress"}}; } void recomputeNextId() { int mx = 1000; for (const auto &r : records) mx = max(mx, r.id); nextId = mx + 1; } public: FeedbackManager() { loadFromFile(); } // File format, one record per line: id|category|rating|status|message void loadFromFile() { records.clear(); ifstream in(DATA_FILE); if (!in) { // first run -> demo data records = seed(); recomputeNextId(); saveToFile(); return; } string ln; while (getline(in, ln)) { vector parts; size_t start = 0; for (int i = 0; i < 4; ++i) { // 4 separators, message is the rest size_t p = ln.find('|', start); if (p == string::npos) break; parts.push_back(ln.substr(start, p - start)); start = p + 1; } if (parts.size() != 4) continue; // skip corrupt line try { Feedback f{stoi(parts[0]), parts[1], ln.substr(start), stoi(parts[2]), parts[3]}; records.push_back(f); } catch (...) {} } recomputeNextId(); } void saveToFile() const { ofstream out(DATA_FILE, ios::trunc); for (const auto &r : records) out << r.id << '|' << r.category << '|' << r.rating << '|' << r.status << '|' << r.message << '\n'; } void resetDemoData() { records = seed(); recomputeNextId(); saveToFile(); } int addFeedback(const string &category, string message, int rating) { replace(message.begin(), message.end(), '|', '/'); // keep file format safe Feedback f{nextId++, category, message, rating, "Pending"}; records.push_back(f); saveToFile(); return f.id; } const Feedback *findById(int id) const { for (const auto &r : records) if (r.id == id) return &r; return nullptr; } bool updateStatus(int id, const string &status) { for (auto &r : records) { if (r.id == id) { r.status = status; saveToFile(); return true; } } return false; } const vector &all() const { return records; } vector filter(const string &category, const string &status, const string &query) const { vector out; string q = util::lower(query); for (const auto &r : records) { if (!category.empty() && r.category != category) continue; if (!status.empty() && r.status != status) continue; if (!q.empty() && util::lower(r.message).find(q) == string::npos && to_string(r.id).find(q) == string::npos) continue; out.push_back(r); } sort(out.begin(), out.end(), [](const Feedback &a, const Feedback &b) { return a.id > b.id; }); return out; } }; // ---------------------------------------------------------------- reports class ReportGenerator { const vector &R; public: explicit ReportGenerator(const vector &records) : R(records) {} vector> categoryCounts() const { vector> out; for (const auto &c : CATEGORIES) { int n = (int)count_if(R.begin(), R.end(), [&](const Feedback &f) { return f.category == c; }); out.push_back({c, n}); } return out; } int statusCount(const string &s) const { return (int)count_if(R.begin(), R.end(), [&](const Feedback &f) { return f.status == s; }); } double averageRating() const { // ignores skipped (0) ratings; -1 if none int sum = 0, n = 0; for (const auto &f : R) if (f.rating > 0) { sum += f.rating; ++n; } return n ? (double)sum / n : -1.0; } // FR9: keywords appearing in 2+ different records vector>> repeatedKeywords(size_t top = 6) const { static const set STOP = { "the", "a", "an", "is", "are", "was", "were", "in", "on", "of", "for", "to", "and", "or", "not", "has", "have", "had", "be", "been", "it", "its", "this", "that", "with", "very", "after", "before", "near", "more", "should", "please", "often", "keeps", "class", "from", "at", "by", "as", "our", "we", "i", "my"}; map> kw; for (const auto &f : R) { string cleaned = util::lower(f.message); for (char &c : cleaned) if (!isalpha(static_cast(c)) && c != '-') c = ' '; istringstream ss(cleaned); string w; set seen; // count once per record while (ss >> w) if (w.size() > 2 && !STOP.count(w) && seen.insert(w).second) kw[w].push_back(f.id); } vector>> out; for (auto &p : kw) if (p.second.size() >= 2) out.push_back(p); sort(out.begin(), out.end(), [](const auto &a, const auto &b) { return a.second.size() != b.second.size() ? a.second.size() > b.second.size() : a.first < b.first; }); if (out.size() > top) out.resize(top); return out; } void print() const { int total = (int)R.size(); double avg = averageRating(); util::line('='); cout << " FEEDBACK REPORT\n"; util::line('='); cout << "Total feedback : " << total << "\n"; cout << "Pending : " << statusCount("Pending") << "\n"; cout << "In Progress : " << statusCount("In Progress") << "\n"; cout << "Resolved : " << statusCount("Resolved") << "\n"; cout << "Average rating : "; if (avg < 0) cout << "-\n"; else cout << fixed << setprecision(1) << avg << " / 5\n"; cout << "\nComplaints per category:\n"; auto cc = categoryCounts(); int mx = 0; for (auto &p : cc) mx = max(mx, p.second); for (auto &p : cc) { cout << " " << left << setw(15) << p.first << right << setw(3) << p.second << " " << string(p.second, '#'); if (p.second == mx && mx > 0) cout << " <-- busiest"; cout << "\n"; } cout << "\nRepeated complaints (keyword in 2+ records):\n"; auto reps = repeatedKeywords(); if (reps.empty()) cout << " No repeated issues yet.\n"; for (auto &p : reps) { cout << " " << left << setw(14) << p.first << right << setw(2) << p.second.size() << "x ["; for (size_t i = 0; i < p.second.size(); ++i) cout << (i ? ", #" : "#") << p.second[i]; cout << "]\n"; } util::line('='); } }; // ---------------------------------------------------------------- UI void printTable(const vector &rows) { if (rows.empty()) { cout << " No feedback matches.\n"; return; } util::line(); cout << left << setw(7) << "ID" << setw(16) << "Category" << setw(46) << "Message" << setw(11) << "Rating" << "Status\n"; util::line(); for (const auto &r : rows) cout << left << setw(7) << r.id << setw(16) << r.category << setw(46) << util::clip(r.message, 44) << setw(11) << util::stars(r.rating) << r.status << "\n"; util::line(); cout << rows.size() << " record(s) shown.\n"; } string pickCategory() { cout << "\nCategories:\n"; for (size_t i = 0; i < CATEGORIES.size(); ++i) cout << " " << i + 1 << ". " << CATEGORIES[i] << "\n"; return CATEGORIES[util::readInt("Choose category (1-6): ", 1, (int)CATEGORIES.size()) - 1]; } string pickStatus() { cout << "\nStatuses:\n"; for (size_t i = 0; i < STATUSES.size(); ++i) cout << " " << i + 1 << ". " << STATUSES[i] << "\n"; return STATUSES[util::readInt("Choose status (1-3): ", 1, (int)STATUSES.size()) - 1]; } void submitFeedback(FeedbackManager &m) { cout << "\n--- Submit Feedback (anonymous) ---\n"; string cat = pickCategory(); string msg; while (true) { msg = util::readLine("Write your message (max 500 chars): "); if (msg.empty()) { cout << " Error: message cannot be empty.\n"; continue; } if (msg.size() > MAX_MSG) { cout << " Error: message too long (" << msg.size() << "/500).\n"; continue; } break; } int rating = util::readInt("Rate your experience 1-5 (0 = skip): ", 0, 5); int id = m.addFeedback(cat, msg, rating); cout << "\n Feedback saved anonymously. Your ID: " << id << " [status = Pending]\n" << " Keep this ID - it is the only way to track your complaint.\n"; } void trackFeedback(const FeedbackManager &m) { cout << "\n--- Track Feedback ---\n"; int id = util::readInt("Enter feedback ID: ", 1, 999999999); const Feedback *f = m.findById(id); if (!f) { cout << " Feedback ID not found.\n"; return; } cout << "\n ID : #" << f->id << "\n Category : " << f->category << "\n Message : " << f->message << "\n Rating : " << util::stars(f->rating) << "\n Status : " << f->status << "\n Progress : "; int idx = (int)(find(STATUSES.begin(), STATUSES.end(), f->status) - STATUSES.begin()); for (int i = 0; i < (int)STATUSES.size(); ++i) cout << (i <= idx ? "[x] " : "[ ] ") << STATUSES[i] << (i + 1 < (int)STATUSES.size() ? " -> " : "\n"); } void adminMenu(FeedbackManager &m) { while (true) { cout << "\n===== ADMIN PANEL =====\n" "1.
